cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PS_VERSION LIMIT_COUNT 1)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/product.py"
  --generate "${CMAKE_CURRENT_BINARY_DIR}/generated" RESULT_VARIABLE PS_PRODUCT_RESULT)
if(NOT PS_PRODUCT_RESULT EQUAL 0)
  message(FATAL_ERROR "Product identity generation failed")
endif()
include("${CMAKE_CURRENT_BINARY_DIR}/generated/product.cmake")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS product.json scripts/product.py)
project(${PS_IDENTIFIER} VERSION ${PS_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PS_LTO "Optimize release builds across module boundaries" ON)
if(PS_LTO AND CMAKE_BUILD_TYPE STREQUAL "Release")
  set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()
option(PS_SANITIZE "Enable address and undefined behavior checks" OFF)
if(PS_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
set(DS vendor/DaisySP/Source)
add_library(daisy STATIC ${DS}/Synthesis/oscillator.cpp ${DS}/Control/adsr.cpp
  ${DS}/PhysicalModeling/stringvoice.cpp ${DS}/PhysicalModeling/KarplusString.cpp
  ${DS}/Dynamics/crossfade.cpp ${DS}/Utility/dcblock.cpp ${DS}/Filters/svf.cpp)
target_include_directories(daisy SYSTEM PUBLIC ${DS} ${DS}/Utility ${DS}/PhysicalModeling)
find_package(Threads REQUIRED)
add_library(instrument STATIC src/music.cpp src/sound_catalog.cpp src/melody_voice.cpp src/controller.cpp src/pitch_gesture.cpp src/guitar_controller.cpp src/engine.cpp src/string_level.cpp src/settings.cpp src/atomic_file.cpp src/drums.cpp src/loop_controls.cpp src/looper.cpp src/loop_pickup.cpp src/loop_view.cpp src/loop_presentation.cpp src/loop_file.cpp src/studio.cpp src/loop_demo.cpp src/loop_cache.cpp)
target_include_directories(instrument PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(instrument PUBLIC daisy Threads::Threads)
target_compile_options(instrument PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CROSSCOMPILING)
  find_path(SDL_INCLUDE_DIR SDL.h PATH_SUFFIXES SDL2 REQUIRED)
  find_library(SDL_LIBRARY SDL2 REQUIRED)
  add_library(SDL2::SDL2 UNKNOWN IMPORTED)
  set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_LOCATION "${SDL_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${SDL_INCLUDE_DIR}")
else()
  find_package(SDL2 REQUIRED CONFIG)
endif()
add_executable(${PS_BINARY} src/main.cpp src/input.cpp src/ui.cpp src/guitar_ui.cpp src/loop_ui.cpp src/ui_controls.cpp src/font.cpp src/notation.cpp src/audio.cpp src/demo.cpp src/sound_demo.cpp src/wave_file.cpp)
target_link_libraries(${PS_BINARY} PRIVATE instrument SDL2::SDL2)
target_compile_options(${PS_BINARY} PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(behavior-tests tests/behavior.cpp)
target_link_libraries(behavior-tests PRIVATE instrument)
add_executable(cache-tests tests/cache_behavior.cpp)
target_link_libraries(cache-tests PRIVATE instrument)
add_test(NAME cached-loops COMMAND cache-tests)
set_tests_properties(cached-loops PROPERTIES TIMEOUT 30)
add_executable(guitar-tests tests/guitar_behavior.cpp)
target_link_libraries(guitar-tests PRIVATE instrument)
add_test(NAME guitar-picking COMMAND guitar-tests)
add_executable(recording-entry-tests tests/recording_entry.cpp)
target_link_libraries(recording-entry-tests PRIVATE instrument)
add_test(NAME recording-entry COMMAND recording-entry-tests)
add_executable(gesture-tests tests/pitch_gesture_behavior.cpp)
target_link_libraries(gesture-tests PRIVATE instrument)
add_test(NAME pitch-gestures COMMAND gesture-tests)
add_executable(pitch-tests tests/pitch_behavior.cpp)
target_link_libraries(pitch-tests PRIVATE instrument)
add_test(NAME attack-pitch COMMAND pitch-tests)
add_executable(sound-tests tests/sound_behavior.cpp)
target_link_libraries(sound-tests PRIVATE instrument)
add_test(NAME sounds COMMAND sound-tests)
add_executable(loop-experience-tests tests/loop_experience.cpp)
target_link_libraries(loop-experience-tests PRIVATE instrument)
add_test(NAME loop-experience COMMAND loop-experience-tests)
add_executable(loop-tests tests/loop_behavior.cpp)
target_link_libraries(loop-tests PRIVATE instrument)
add_test(NAME loops-and-drums COMMAND loop-tests)
add_executable(audit-tests tests/audit.cpp)
target_link_libraries(audit-tests PRIVATE instrument)
add_executable(audio-tests tests/audio_behavior.cpp src/audio.cpp)
target_link_libraries(audio-tests PRIVATE instrument SDL2::SDL2)
add_test(NAME audio-queue COMMAND audio-tests)
add_test(NAME audit COMMAND audit-tests)
add_test(NAME behavior COMMAND behavior-tests)
add_test(NAME app-smoke COMMAND ${PS_BINARY} --smoke)
add_test(NAME app-loop-smoke COMMAND ${PS_BINARY} --smoke --loop-demo --seconds .6)
set_tests_properties(app-smoke app-loop-smoke PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
add_executable(performance-measurement EXCLUDE_FROM_ALL tests/measure_performance.cpp)
target_link_libraries(performance-measurement PRIVATE instrument)
add_executable(level-measurement EXCLUDE_FROM_ALL tests/measure_loudness.cpp)
target_link_libraries(level-measurement PRIVATE instrument)
find_package(Python3 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME app-cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/app_cli.py" "$<TARGET_FILE:${PS_BINARY}>")
endif()
install(TARGETS ${PS_BINARY} RUNTIME DESTINATION bin)
